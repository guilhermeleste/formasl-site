examples/F02-nary-semantic-roles/canonical.fsl
module semantic_roles {
    type Agent
    type Artifact
    type Project
    type Environment

    relation Assignment {
        role assignee: Agent
        role artifact: Artifact
        role project: Project
        role environment: Environment
    }

    assert claim a1 : Assignment(
        assignee: $alice,
        artifact: $dataset,
        project: $atlas,
        environment: $production
    ) { by : $registry }
}
概念演示

四个角色，一个关系。

Assignment 将四个具有类型的参与者纳入同一关系实例。Claim 记录可归属的断言行为。

module query_basic {
    type Artifact
    relation DependsOn {
        role dependent: Artifact
        role prerequisite: Artifact
    }

    assert claim d1 : DependsOn(
        dependent: $app,
        prerequisite: $parser
    ) { by : $manifest }

    match DirectDependencies() {
        pattern ?d: DependsOn(
            dependent: ?dependent,
            prerequisite: ?prerequisite
        )
    }
}
概念演示

查询而不修改源数据。

examples/F05-reasoning-completion/canonical.fsl
rule Base using DependsOn(?x: Artifact) {
    when DependsOn(dependent: ?a, prerequisite: ?b)
    derive positive ReachableDependency(
        dependent: ?a, prerequisite: ?b
    )
}

rule Step using DependsOn(?x: Artifact) {
    when DependsOn(dependent: ?a, prerequisite: ?b)
    when ReachableDependency(dependent: ?b, prerequisite: ?c)
    derive positive ReachableDependency(
        dependent: ?a, prerequisite: ?c
    )
}
概念演示

推导不是断言。

规则给出推导结论；推理本身不自动产生可归属的断言。

transform Normalize($d: Device) using Reading {
    rule normalize {
        match {
            pattern ?r: Reading(

            role value: ?value

03 / FormaSL 操作族FORMA

不同操作具有不同含义

match 查询关系；rule 与 infer 推导结论；analyze 执行分析和模型计算；transform 构造新结构；validate、admit、authorize 和 decide 管理规则与责任；scenario 和 composition 用于假设评估及受控组合。

type / relation

定义类别及关系中参与者的角色。

assert claim

记录可识别、可归属的断言。

match

查询模式、路径和聚合结果。

rule / infer

依据声明的规则与理论推导结论。

analyze

运行分析、统计与模型，包括图算法及图神经网络。

transform

按照声明的映射关系构建结构。

validate / admit

检查条件并控制结果准入。

authorize / decide

区分授权与承担责任的决策。

scenario

评估假设情境。

composition

组合具有明确依赖关系的操作。
